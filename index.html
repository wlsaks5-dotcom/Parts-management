let html5QrCodeEdit = null; // 수정 모달에서 QR 스캔 시작 function startQrScannerForEdit() { const container = document.getElementById('edit-qr-container'); container.style.display = 'block'; if (!html5QrCodeEdit) { html5QrCodeEdit = new Html5Qrcode("edit-reader"); } const config = { fps: 10, qrbox: { width: 200, height: 200 } }; html5QrCodeEdit.start( { facingMode: "environment" }, config, (decodedText, decodedResult) => { document.getElementById('modalEditPartQrCode').value = decodedText; alert(`QR 스캔 완료: ${decodedText}`); stopQrScannerForEdit(); }, (errorMessage) => {} ).catch(err => { console.error("카메라 시작 실패:", err); alert("카메라 권한을 확인해주세요."); container.style.display = 'none'; }); } // 수정 모달 QR 스캔 중지 function stopQrScannerForEdit() { const container = document.getElementById('edit-qr-container'); if (html5QrCodeEdit) { html5QrCodeEdit.stop().then(() => { container.style.display = 'none'; }).catch(err => { container.style.display = 'none'; }); } else { container.style.display = 'none'; } } // 모달 닫을 때 카메라도 같이 끄기 function closeAdminPartEditModal() { stopQrScannerForEdit(); document.getElementById('adminPartEditModal').style.display = 'none'; } // 수정 사항 저장 함수 수정 async function saveAdminPartEdit() { const id = document.getElementById('editDocId').value; const name = document.getElementById('modalEditPartName').value.trim(); const stock = parseInt(document.getElementById('modalEditPartStock').value) || 0; const type = document.getElementById('modalEditPartType').value; const qrCode = document.getElementById('modalEditPartQrCode').value.trim(); if (!name) { alert("부품 이름을 입력해주세요."); return; } try { await db.collection("parts").doc(id).update({ name, stock, type, qrCode, // QR 정보 함께 저장 updatedAt: firebase.firestore.FieldValue.serverTimestamp() }); alert("부품 정보 및 QR 연동이 수정되었습니다."); closeAdminPartEditModal(); loadRegisteredPartsList(); preloadPartsCache(); } else { // ... (에러 처리) } }
